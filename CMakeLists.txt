cmake_minimum_required(VERSION 3.22)
project(TalkTranscriber VERSION 0.1.0 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

option(TALK_TRANSCRIBER_BUILD_PLUGIN "Build the JUCE VST3 plugin" OFF)
option(TALK_TRANSCRIBER_BUILD_SERVER "Build the native local server" ON)
option(TALK_TRANSCRIBER_BUILD_TESTS "Build protocol and VAD tests" ON)
set(WHISPER_DIR "" CACHE PATH "Path to whisper.cpp checkout")
set(IXWEBSOCKET_DIR "" CACHE PATH "Path to IXWebSocket checkout")

if(TALK_TRANSCRIBER_BUILD_SERVER)
  if(IXWEBSOCKET_DIR)
    set(USE_TLS OFF CACHE BOOL "" FORCE)
    add_subdirectory(${IXWEBSOCKET_DIR} ${CMAKE_BINARY_DIR}/ixwebsocket)
    set(TALK_TRANSCRIBER_HAS_WEBSOCKET ON)
  endif()
  if(WHISPER_DIR)
    set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
    set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
    set(GGML_METAL ON CACHE BOOL "" FORCE)
    add_subdirectory(${WHISPER_DIR} ${CMAKE_BINARY_DIR}/whisper)
    set(TALK_TRANSCRIBER_HAS_WHISPER ON)
  endif()
  add_subdirectory(server)
endif()

if(TALK_TRANSCRIBER_BUILD_PLUGIN)
  add_subdirectory(plugin)
endif()

if(TALK_TRANSCRIBER_BUILD_TESTS)
  enable_testing()
  add_subdirectory(tests)
endif()
